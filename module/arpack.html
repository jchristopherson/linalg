interface

  • public subroutine dnaupd(ido, bmat, n, which, nev, tol, resid, ncv, v, ldv, iparam, ipntr, workd, workl, lworkl, info)

    Implements the implicitly restarted Arnoldi iteration for non-symmetric problems by means of a reverse communication interface.

    Arguments

    Type IntentOptional Attributes Name
    integer(kind=int32), intent(inout) :: ido

    The reverse communication flag.
    character(len=1), intent(in) :: bmat

    'I' for a standard problem, or 'G' for a generalized problem.
    integer(kind=int32), intent(in) :: n

    The dimension of the problem.
    character(len=2), intent(in) :: which

    Identifies which eigenvalues to compute ('LM', 'SM', 'LR', 'SR', 'LI', or 'SI').
    integer(kind=int32), intent(inout) :: nev

    The number of eigenvalues to compute.
    real(kind=real64), intent(inout) :: tol

    The convergence tolerance. If zero, machine precision is used.
    real(kind=real64), intent(inout) :: resid(*)

    An N-element residual vector.
    integer(kind=int32), intent(in) :: ncv

    The number of Arnoldi basis vectors to use.
    real(kind=real64), intent(inout) :: v(ldv,*)

    An LDV-by-NCV matrix containing the Arnoldi basis vectors.
    integer(kind=int32), intent(in) :: ldv

    The leading dimension of V.
    integer(kind=int32), intent(inout) :: iparam(*)

    An 11-element array containing the iteration parameters.
    integer(kind=int32), intent(inout) :: ipntr(*)

    A 14-element array containing pointers into WORKD and WORKL.
    real(kind=real64), intent(inout) :: workd(*)

    A 3*N-element reverse communication workspace array.
    real(kind=real64), intent(inout) :: workl(*)

    An LWORKL-element workspace array.
    integer(kind=int32), intent(in) :: lworkl

    The length of WORKL. Must be at least 3 * NCV**2 + 6 * NCV.
    integer(kind=int32), intent(inout) :: info

    The status flag.

    Extracts the eigenvalues and, optionally, the eigenvectors, from the results of DNAUPD.

    An NCV-element array used when HOWMNY is 'S'. The array is also used as workspace when HOWMNY is 'A'.
    real(kind=real64), intent(out) :: dr(*)

    An (NEV+1)-element array containing the real components of the eigenvalues.
    real(kind=real64), intent(out) :: di(*)

    An (NEV+1)-element array containing the imaginary components of the eigenvalues.
    real(kind=real64), intent(out) :: z(ldz,*)

    An LDZ-by-(NEV+1) matrix containing the eigenvectors stored in the compact form used by LAPACK's DGEEV.
    integer(kind=int32), intent(in) :: ldz

    The leading dimension of Z.
    real(kind=real64), intent(in) :: sigmar

    The real component of the shift used in shift-invert modes.
    real(kind=real64), intent(in) :: sigmai

    The imaginary component of the shift used in shift-invert modes.
    real(kind=real64), intent(out) :: workev(*)

    A 3*NCV-element workspace array.
    character(len=1), intent(in) :: bmat

    'I' for a standard problem, or 'G' for a generalized problem.
    integer(kind=int32), intent(in) :: n

    The dimension of the problem.
    character(len=2), intent(in) :: which

    Identifies which eigenvalues were computed.
    integer(kind=int32), intent(inout) :: nev

    The number of eigenvalues requested.
    real(kind=real64), intent(inout) :: tol

    The convergence tolerance.
    real(kind=real64), intent(inout) :: resid(*)

    The N-element residual vector from DNAUPD.
    integer(kind=int32), intent(in) :: ncv

    The number of Arnoldi basis vectors used.
    real(kind=real64), intent(inout) :: v(ldv,*)

    The LDV-by-NCV Arnoldi basis from DNAUPD.
    integer(kind=int32), intent(in) :: ldv

    The leading dimension of V.
    integer(kind=int32), intent(inout) :: iparam(*)

    The 11-element iteration parameter array from DNAUPD.
    integer(kind=int32), intent(inout) :: ipntr(*)

    The 14-element pointer array from DNAUPD.
    real(kind=real64), intent(inout) :: workd(*)

    The reverse communication workspace array from DNAUPD.
    real(kind=real64), intent(inout) :: workl(*)

    The LWORKL-element workspace array from DNAUPD.
    integer(kind=int32), intent(in) :: lworkl

    The length of WORKL.
    integer(kind=int32), intent(inout) :: info

    The status flag.
